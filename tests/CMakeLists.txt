function(gc_add_test name source)
	add_executable(genicam-${name} "${source}")
	target_include_directories(genicam-${name} PRIVATE ../src)
	target_link_libraries(genicam-${name} PRIVATE ${ARGN})
	gc_warnings(genicam-${name})
	add_test(NAME genicam-${name} COMMAND genicam-${name})
	set_tests_properties(genicam-${name} PROPERTIES TIMEOUT 15)
endfunction()

gc_add_test(gendc unit/gendc.c genicam::common)
if(GC_BUILD_CAMERA)
	gc_add_test(camera unit/camera.c genicam::camera)
endif()
if(GC_BUILD_HOST AND GC_BUILD_CAMERA)
	gc_add_test(integration integration/gige.c genicam::host genicam::camera ZLIB::ZLIB)
endif()
